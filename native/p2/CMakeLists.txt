cmake_minimum_required(VERSION 3.25)
project(CatalystRepastProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REPAST_SOURCE "" CACHE PATH "Pinned repast.hpc checkout")
if(NOT EXISTS "${REPAST_SOURCE}/src/repast_hpc/RepastProcess.cpp")
  message(FATAL_ERROR "Set REPAST_SOURCE to the pinned upstream checkout")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND ${GIT_EXECUTABLE} -C "${REPAST_SOURCE}" rev-parse HEAD
  OUTPUT_VARIABLE repast_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT repast_revision STREQUAL "a96c15f99589d15352523a0a710362af8997852a")
  message(FATAL_ERROR "P2 requires the qualified Repast HPC revision")
endif()
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(Boost 1.92.0 EXACT CONFIG REQUIRED COMPONENTS serialization filesystem)
# Homebrew splits MPI into another prefix; its config references a helper that
# only exists in the main Boost prefix. Link the version-matched library directly.
find_library(BOOST_MPI_LIBRARY NAMES boost_mpi HINTS /opt/homebrew/opt/boost-mpi/lib REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
# Repast uses C++11-era function base classes supplied by libc++ compatibility.
add_compile_definitions(_LIBCPP_ENABLE_CXX17_REMOVED_UNARY_BINARY_FUNCTION)
file(GLOB repast_sources CONFIGURE_DEPENDS "${REPAST_SOURCE}/src/repast_hpc/*.cpp")
# NetCDF recorders are outside P2; actual process, scheduler and migration remain.
list(FILTER repast_sources EXCLUDE REGEX "/NCDataSet(Builder)?\\.cpp$")
add_library(repast_probe_core STATIC ${repast_sources})
target_include_directories(repast_probe_core PUBLIC "${REPAST_SOURCE}/src")
target_link_libraries(repast_probe_core PUBLIC MPI::MPI_CXX ${BOOST_MPI_LIBRARY} Boost::serialization Boost::filesystem)
target_compile_options(repast_probe_core PRIVATE -Wno-deprecated-declarations)
add_executable(catalyst_repast main.cpp)
target_link_libraries(catalyst_repast PRIVATE repast_probe_core OpenSSL::Crypto)
target_compile_options(catalyst_repast PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
add_executable(catalyst_value_tests tests.cpp)
target_link_libraries(catalyst_value_tests PRIVATE Boost::serialization OpenSSL::Crypto)
enable_testing()
add_test(NAME canonical_and_random COMMAND catalyst_value_tests)
